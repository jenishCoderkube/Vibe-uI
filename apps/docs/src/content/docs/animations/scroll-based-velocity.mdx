# Scroll Based Velocity

A responsive text marquee whose velocity and scroll-direction react to the user's scroll.

## Installation

<Installation name="scroll-based-velocity" npmDeps="motion">

```tsx
"use client"

import React, { useContext, useEffect, useRef, useState } from "react"
import {
  motion,
  useAnimationFrame,
  useMotionValue,
  useScroll,
  useSpring,
  useTransform,
  useVelocity,
} from "motion/react"
import type { MotionValue } from "motion/react"

import { cn } from "@/lib/utils"

export interface ScrollVelocityRowProps extends React.HTMLAttributes<HTMLDivElement> {
  children: React.ReactNode
  baseVelocity: number
  direction?: 1 | -1
  scrollReactivity?: boolean
}

const ScrollVelocityContext = React.createContext<{
  scrollY: MotionValue<number>
  scrollVelocity: MotionValue<number>
  smoothVelocity: MotionValue<number>
  velocityFactor: MotionValue<number>
} | null>(null)

export function ScrollVelocityContainer({
  children,
  ...props
}: React.HTMLAttributes<HTMLDivElement>) {
  const { scrollY } = useScroll()
  const scrollVelocity = useVelocity(scrollY)
  const smoothVelocity = useSpring(scrollVelocity, {
    damping: 50,
    stiffness: 400,
  })
  const velocityFactor = useTransform(smoothVelocity, [0, 1000], [0, 5], {
    clamp: false,
  })

  return (
    <ScrollVelocityContext.Provider
      value={{ scrollY, scrollVelocity, smoothVelocity, velocityFactor }}
    >
      <div {...props}>{children}</div>
    </ScrollVelocityContext.Provider>
  )
}

function wrap(min: number, max: number, v: number) {
  const rangeSize = max - min
  return ((((v - min) % rangeSize) + rangeSize) % rangeSize) + min
}

interface ScrollVelocityRowImplProps extends ScrollVelocityRowProps {
  velocityFactor: MotionValue<number>
}

function ScrollVelocityRowImpl({
  children,
  baseVelocity = 5,
  direction = 1,
  scrollReactivity = true,
  className,
  ...props
}: ScrollVelocityRowImplProps) {
  const baseX = useMotionValue(0)
  const { velocityFactor } = useContext(ScrollVelocityContext)!

  const containerRef = useRef<HTMLDivElement>(null)
  const blockRef = useRef<HTMLDivElement>(null)

  const [unitWidth, setUnitWidth] = useState(0)
  const [numCopies, setNumCopies] = useState(2)

  const isPageVisibleRef = useRef(true)
  const isInViewRef = useRef(true)
  const prefersReducedMotionRef = useRef(false)

  const baseDirectionRef = useRef(direction)
  const currentDirectionRef = useRef(direction)

  useEffect(() => {
    const handleVisibility = () => {
      isPageVisibleRef.current = document.visibilityState === "visible"
    }

    const updateSizes = () => {
      if (!containerRef.current || !blockRef.current) return
      const containerWidth = containerRef.current.offsetWidth
      const blockWidth = blockRef.current.offsetWidth
      if (blockWidth > 0) {
        setUnitWidth(blockWidth)
        const needed = Math.ceil(containerWidth / blockWidth) + 2
        setNumCopies(needed)
      }
    }

    const handlePRM = (e: MediaQueryListEvent | MediaQueryList) => {
      prefersReducedMotionRef.current = e.matches
    }

    let ro: ResizeObserver | null = null
    let io: IntersectionObserver | null = null
    let mq: MediaQueryList | null = null

    const container = containerRef.current
    const block = blockRef.current

    if (container && block) {
      updateSizes()

      ro = new ResizeObserver(updateSizes)
      ro.observe(container)
      ro.observe(block)

      io = new IntersectionObserver(([entry]) => {
        isInViewRef.current = entry.isIntersecting
      })
      io.observe(container)

      document.addEventListener("visibilitychange", handleVisibility, {
        passive: true,
      })
      handleVisibility()

      mq = window.matchMedia("(prefers-reduced-motion: reduce)")
      mq.addEventListener("change", handlePRM)
      handlePRM(mq)
    }

    return () => {
      if (ro) ro.disconnect()
      if (io) io.disconnect()
      document.removeEventListener("visibilitychange", handleVisibility)
      if (mq) mq.removeEventListener("change", handlePRM)
    }
  }, [])

  const x = useTransform(baseX, (v) => `${-wrap(0, unitWidth, v)}px`)

  useAnimationFrame((_, delta) => {
    if (!isInViewRef.current || !isPageVisibleRef.current || unitWidth === 0)
      return

    const dt = delta / 1000
    const vf = scrollReactivity ? velocityFactor.get() : 0
    const absVf = Math.min(5, Math.abs(vf))
    const speedMultiplier = prefersReducedMotionRef.current ? 1 : 1 + absVf

    if (absVf > 0.1) {
      currentDirectionRef.current = baseDirectionRef.current * (vf >= 0 ? 1 : -1)
    }

    const moveBy =
      currentDirectionRef.current *
      ((unitWidth * baseVelocity) / 100) *
      speedMultiplier *
      dt
    baseX.set(baseX.get() + moveBy)
  })

  return (
    <div
      ref={containerRef}
      className={cn("overflow-hidden whitespace-nowrap", className)}
      {...props}
    >
      <motion.div
        className="inline-flex transform-gpu items-center will-change-transform"
        style={{ x }}
      >
        {Array.from({ length: numCopies }).map((_, i) => (
          <div
            key={i}
            ref={i === 0 ? blockRef : null}
            className="inline-flex items-center"
          >
            {children}
          </div>
        ))}
      </motion.div>
    </div>
  )
}

export function ScrollVelocityRow(props: ScrollVelocityRowProps) {
  const context = useContext(ScrollVelocityContext)
  if (context) {
    return <ScrollVelocityRowImpl {...props} velocityFactor={context.velocityFactor} />
  }

  const { scrollY } = useScroll()
  const scrollVelocity = useVelocity(scrollY)
  const smoothVelocity = useSpring(scrollVelocity, {
    damping: 50,
    stiffness: 400,
  })
  const velocityFactor = useTransform(smoothVelocity, [0, 1000], [0, 5], {
    clamp: false,
  })

  return <ScrollVelocityRowImpl {...props} velocityFactor={velocityFactor} />
}
```
