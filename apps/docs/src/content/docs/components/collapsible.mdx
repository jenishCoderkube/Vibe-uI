# Collapsible

An interactive container component that toggles vertical visibility of content sections with smooth animations.

<ComponentPreview>
  <CollapsibleBasic />

```tsx
import React, { useState } from 'react'
import {
  Collapsible,
  CollapsibleTrigger,
  CollapsibleContent,
} from '@/components/ui/collapsible'
import { Button } from '@/components/ui/button'
import { ChevronsUpDown, Settings } from 'lucide-react'

export default function CollapsibleDemo() {
  const [open, setOpen] = useState(true)

  return (
    <div className="w-full max-w-[350px] rounded-xl border border-border bg-card p-4 shadow-sm text-card-foreground">
      <Collapsible open={open} onOpenChange={setOpen} className="space-y-3">
        <div className="flex items-center justify-between gap-4">
          <div className="flex items-center gap-2">
            <Settings className="h-4 w-4 text-primary" />
            <span className="text-xs font-bold uppercase tracking-wider">
              System Preferences
            </span>
          </div>
          <CollapsibleTrigger asChild>
            <Button variant="ghost" size="icon" className="h-7 w-7">
              <ChevronsUpDown className="h-4 w-4" />
            </Button>
          </CollapsibleTrigger>
        </div>

        <div className="rounded-lg border border-border bg-muted/40 px-3.5 py-2.5 text-xs text-muted-foreground">
          This setting section is permanently active on the machine.
        </div>

        <CollapsibleContent className="space-y-2">
          <div className="rounded-lg border border-border bg-muted/40 px-3.5 py-2.5 text-xs text-muted-foreground">
            Advanced preferences and layout customization options slide open when toggled.
          </div>
        </CollapsibleContent>
      </Collapsible>
    </div>
  )
}
```

```tsx
// @/components/ui/collapsible.tsx
"use client"

import * as React from "react"
import * as CollapsiblePrimitive from "@radix-ui/react-collapsible"

function Collapsible({
  ...props
}: React.ComponentProps<typeof CollapsiblePrimitive.Root>) {
  return <CollapsiblePrimitive.Root data-slot="collapsible" {...props} />
}

function CollapsibleTrigger({
  ...props
}: React.ComponentProps<typeof CollapsiblePrimitive.CollapsibleTrigger>) {
  return (
    <CollapsiblePrimitive.CollapsibleTrigger
      data-slot="collapsible-trigger"
      {...props}
    />
  )
}

function CollapsibleContent({
  ...props
}: React.ComponentProps<typeof CollapsiblePrimitive.CollapsibleContent>) {
  return (
    <CollapsiblePrimitive.CollapsibleContent
      data-slot="collapsible-content"
      {...props}
    />
  )
}

export { Collapsible, CollapsibleTrigger, CollapsibleContent }
```

## Directory Tree

An interactive directory structure representing a real-world IDE workspace tree using nested collapsibles.

<ComponentPreview>
  <CollapsibleCode />

```tsx
import React, { useState } from 'react'
import {
  Collapsible,
  CollapsibleTrigger,
  CollapsibleContent,
} from '@/components/ui/collapsible'
import { Button } from '@/components/ui/button'
import {
  ChevronRight,
  Folder,
  FolderOpen,
  FileCode2,
  FileText,
  Settings,
} from 'lucide-react'

export default function WorkspaceTreeDemo() {
  const [appsOpen, setAppsOpen] = useState(true)
  const [docsOpen, setDocsOpen] = useState(true)
  const [packagesOpen, setPackagesOpen] = useState(true)
  const [uiOpen, setUiOpen] = useState(true)
  const [srcOpen, setSrcOpen] = useState(true)
  const [componentsOpen, setComponentsOpen] = useState(true)

  return (
    <div className="w-full max-w-[360px] border border-border/60 bg-zinc-950/45 text-white rounded-xl p-4 shadow-xl font-mono text-xs">
      {/* apps/ folder */}
      <Collapsible open={appsOpen} onOpenChange={setAppsOpen}>
        <div className="flex items-center justify-between py-1 hover:bg-neutral-800/20 rounded px-1.5 cursor-pointer">
          <CollapsibleTrigger asChild>
            <div className="flex items-center gap-2 flex-1">
              <ChevronRight className={`h-3 w-3 text-neutral-500 transition-transform duration-200 ${appsOpen ? 'rotate-90' : ''}`} />
              {appsOpen ? <FolderOpen className="h-3.5 w-3.5 text-blue-400" /> : <Folder className="h-3.5 w-3.5 text-blue-400" />}
              <span className="text-neutral-200">apps</span>
            </div>
          </CollapsibleTrigger>
        </div>
        
        <CollapsibleContent className="pl-3 ml-2.5 border-l border-neutral-800 space-y-1 mt-0.5">
          {/* docs/ folder */}
          <Collapsible open={docsOpen} onOpenChange={setDocsOpen}>
            <div className="flex items-center justify-between py-1 hover:bg-neutral-800/20 rounded px-1.5 cursor-pointer">
              <CollapsibleTrigger asChild>
                <div className="flex items-center gap-2 flex-1">
                  <ChevronRight className={`h-3 w-3 text-neutral-500 transition-transform duration-200 ${docsOpen ? 'rotate-90' : ''}`} />
                  {docsOpen ? <FolderOpen className="h-3.5 w-3.5 text-blue-400" /> : <Folder className="h-3.5 w-3.5 text-blue-400" />}
                  <span className="text-neutral-200">docs</span>
                </div>
              </CollapsibleTrigger>
            </div>
            
            <CollapsibleContent className="pl-3 ml-2.5 border-l border-neutral-800 space-y-1 mt-0.5">
              <div className="flex items-center gap-2 py-1 px-1.5 text-neutral-400 hover:text-white cursor-pointer rounded hover:bg-neutral-800/20">
                <FileCode2 className="h-3.5 w-3.5 text-amber-500" />
                <span>next.config.js</span>
              </div>
              <div className="flex items-center gap-2 py-1 px-1.5 text-neutral-400 hover:text-white cursor-pointer rounded hover:bg-neutral-800/20">
                <FileText className="h-3.5 w-3.5 text-neutral-500" />
                <span>package.json</span>
              </div>
            </CollapsibleContent>
          </Collapsible>
        </CollapsibleContent>
      </Collapsible>

      {/* packages/ folder */}
      <Collapsible open={packagesOpen} onOpenChange={setPackagesOpen}>
        <div className="flex items-center justify-between py-1 hover:bg-neutral-800/20 rounded px-1.5 cursor-pointer">
          <CollapsibleTrigger asChild>
            <div className="flex items-center gap-2 flex-1">
              <ChevronRight className={`h-3 w-3 text-neutral-500 transition-transform duration-200 ${packagesOpen ? 'rotate-90' : ''}`} />
              {packagesOpen ? <FolderOpen className="h-3.5 w-3.5 text-blue-400" /> : <Folder className="h-3.5 w-3.5 text-blue-400" />}
              <span className="text-neutral-200">packages</span>
            </div>
          </CollapsibleTrigger>
        </div>
        
        <CollapsibleContent className="pl-3 ml-2.5 border-l border-neutral-800 space-y-1 mt-0.5">
          {/* ui/ folder */}
          <Collapsible open={uiOpen} onOpenChange={setUiOpen}>
            <div className="flex items-center justify-between py-1 hover:bg-neutral-800/20 rounded px-1.5 cursor-pointer">
              <CollapsibleTrigger asChild>
                <div className="flex items-center gap-2 flex-1">
                  <ChevronRight className={`h-3 w-3 text-neutral-500 transition-transform duration-200 ${uiOpen ? 'rotate-90' : ''}`} />
                  {uiOpen ? <FolderOpen className="h-3.5 w-3.5 text-blue-400" /> : <Folder className="h-3.5 w-3.5 text-blue-400" />}
                  <span className="text-neutral-200">ui</span>
                </div>
              </CollapsibleTrigger>
            </div>
            
            <CollapsibleContent className="pl-3 ml-2.5 border-l border-neutral-800 space-y-1 mt-0.5">
              {/* src/ folder */}
              <Collapsible open={srcOpen} onOpenChange={setSrcOpen}>
                <div className="flex items-center justify-between py-1 hover:bg-neutral-800/20 rounded px-1.5 cursor-pointer">
                  <CollapsibleTrigger asChild>
                    <div className="flex items-center gap-2 flex-1">
                      <ChevronRight className={`h-3 w-3 text-neutral-500 transition-transform duration-200 ${srcOpen ? 'rotate-90' : ''}`} />
                      {srcOpen ? <FolderOpen className="h-3.5 w-3.5 text-blue-400" /> : <Folder className="h-3.5 w-3.5 text-blue-400" />}
                      <span className="text-neutral-200">src</span>
                    </div>
                  </CollapsibleTrigger>
                </div>
                
                <CollapsibleContent className="pl-3 ml-2.5 border-l border-neutral-800 space-y-1 mt-0.5">
                  {/* components/ folder */}
                  <Collapsible open={componentsOpen} onOpenChange={setComponentsOpen}>
                    <div className="flex items-center justify-between py-1 hover:bg-neutral-800/20 rounded px-1.5 cursor-pointer">
                      <CollapsibleTrigger asChild>
                        <div className="flex items-center gap-2 flex-1">
                          <ChevronRight className={`h-3 w-3 text-neutral-500 transition-transform duration-200 ${componentsOpen ? 'rotate-90' : ''}`} />
                          {componentsOpen ? <FolderOpen className="h-3.5 w-3.5 text-blue-400" /> : <Folder className="h-3.5 w-3.5 text-blue-400" />}
                          <span className="text-neutral-200">components</span>
                        </div>
                      </CollapsibleTrigger>
                    </div>
                    
                    <CollapsibleContent className="pl-3 ml-2.5 border-l border-neutral-800 space-y-1 mt-0.5">
                      <div className="flex items-center gap-2 py-1 px-1.5 text-neutral-400 hover:text-white cursor-pointer rounded hover:bg-neutral-800/20">
                        <FileCode2 className="h-3.5 w-3.5 text-emerald-500" />
                        <span>button.tsx</span>
                      </div>
                      <div className="flex items-center gap-2 py-1 px-1.5 text-neutral-400 hover:text-white cursor-pointer rounded hover:bg-neutral-800/20">
                        <FileCode2 className="h-3.5 w-3.5 text-emerald-500" />
                        <span>collapsible.tsx</span>
                      </div>
                    </CollapsibleContent>
                  </Collapsible>
                </CollapsibleContent>
              </Collapsible>
            </CollapsibleContent>
          </Collapsible>
        </CollapsibleContent>
      </Collapsible>

      {/* Root config files */}
      <div className="flex items-center gap-2 py-1 px-1.5 text-neutral-400 hover:text-white cursor-pointer rounded hover:bg-neutral-800/20 pl-7">
        <FileCode2 className="h-3.5 w-3.5 text-purple-500" />
        <span>turbo.json</span>
      </div>
    </div>
  )
}
```
