<Installation name="input" vibeDeps="label">

```tsx
// @/components/ui/input.tsx
import * as React from 'react'
import { cn } from '@/lib/utils'

export interface InputProps extends React.InputHTMLAttributes<HTMLInputElement> {
  error?: boolean
  variant?: 'default' | 'bottom' | 'glass'
}

const Input = React.forwardRef<HTMLInputElement, InputProps>(
  ({ className, type, error, variant = 'default', ...props }, ref) => {
    return (
      <input
        type={type}
        className={cn(
          'flex h-10 w-full px-3 py-2 text-sm transition-all duration-200 focus-visible:outline-none disabled:cursor-not-allowed disabled:opacity-50',
          // Variants
          variant === 'default' &&
            'rounded-md border border-input bg-background ring-offset-background placeholder:text-muted-foreground focus-visible:ring-2 focus-visible:ring-ring focus-visible:ring-offset-2 shadow-sm',
          variant === 'bottom' &&
            'rounded-none border-0 border-b border-input bg-transparent px-0 focus-visible:border-primary shadow-none focus-visible:ring-0',
          variant === 'glass' &&
            'rounded-md border border-white/20 dark:border-white/10 bg-white/5 dark:bg-black/20 backdrop-blur-md placeholder:text-muted-foreground/60 focus-visible:ring-2 focus-visible:ring-white/30 focus-visible:bg-white/10 dark:focus-visible:bg-black/30 shadow-sm',
          // File input overrides
          'file:border-0 file:bg-transparent file:text-sm file:font-medium file:text-foreground',
          // Error states
          error &&
            variant === 'default' &&
            'border-destructive focus-visible:ring-destructive',
          error &&
            variant === 'bottom' &&
            'border-destructive focus-visible:border-destructive',
          error &&
            variant === 'glass' &&
            'border-destructive focus-visible:ring-destructive',
          className,
        )}
        ref={ref}
        {...props}
      />
    )
  },
)
Input.displayName = 'Input'

export interface FloatingInputProps extends React.InputHTMLAttributes<HTMLInputElement> {
  label: string
  error?: boolean
  variant?: 'default' | 'bottom' | 'glass'
}

const FloatingInput = React.forwardRef<HTMLInputElement, FloatingInputProps>(
  ({ className, label, type, error, variant = 'default', id, ...props }, ref) => {
    const inputId = id || React.useId()
    return (
      <div className="relative mt-2">
        <input
          type={type}
          id={inputId}
          placeholder=" "
          className={cn(
            'peer flex h-10 w-full px-3 py-2 text-sm transition-all duration-200 focus-visible:outline-none disabled:cursor-not-allowed disabled:opacity-50 pt-5 pb-1.5',
            // Variants
            variant === 'default' &&
              'rounded-md border border-input bg-background ring-offset-background focus-visible:ring-1 focus-visible:ring-primary focus-visible:border-primary shadow-sm',
            variant === 'bottom' &&
              'rounded-none border-0 border-b border-input bg-transparent px-0 focus-visible:border-primary shadow-none focus-visible:ring-0',
            variant === 'glass' &&
              'rounded-md border border-white/20 dark:border-white/10 bg-white/5 dark:bg-black/20 backdrop-blur-md focus-visible:ring-1 focus-visible:ring-white/30 focus-visible:bg-white/10 dark:focus-visible:bg-black/30 shadow-sm',
            // Error states
            error && 'border-destructive focus-visible:ring-destructive focus-visible:border-destructive',
            className
          )}
          ref={ref}
          {...props}
        />
        <label
          htmlFor={inputId}
          className={cn(
            'absolute text-sm text-muted-foreground duration-300 transform -translate-y-3.5 scale-75 top-3 z-10 origin-[0] pointer-events-none select-none transition-all',
            // Label variant layout
            variant === 'bottom' ? 'left-0' : 'left-3 bg-background px-1',
            variant === 'glass' && 'bg-zinc-950 dark:bg-zinc-950 px-1',
            // Peer states (when focused or has value)
            'peer-placeholder-shown:scale-100 peer-placeholder-shown:translate-y-0',
            'peer-focus:scale-75 peer-focus:-translate-y-3.5 peer-focus:text-primary',
            // Error state
            error && 'text-destructive peer-focus:text-destructive'
          )}
        >
          {label}
        </label>
      </div>
    )
  }
)
FloatingInput.displayName = 'FloatingInput'

export { Input, FloatingInput }
```

</Installation>

## Component Concept

Vibe UI's Input component captures single-line text data. It features customized focus ring styles and visual presets to match the overall design framework of the monorepo.

### Preset Variations

- **Glassmorphic**: Frosted input field backgrounds with transparent borders.
- **Retro**: Thick dark input border outlines with solid background profiles.
- **Neon Glow**: Highlight focus border frames that cast secondary neon drop shadows.
