# Empty State

An empty state container used to present clear guidance, icons, and call-to-action buttons when content is absent.

```tsx
'use client'

import * as React from 'react'
import { tv, type VariantProps } from 'tailwind-variants'
import { cn } from '@/lib/utils'

const emptyVariants = tv({
  base: 'flex flex-col items-center justify-center text-center p-8 rounded-2xl border transition-all duration-200',
  variants: {
    variant: {
      default: 'bg-card border-border shadow-xs text-card-foreground',
      glass:
        'bg-white/5 dark:bg-black/30 border-white/10 dark:border-white/5 backdrop-blur-md shadow-md text-foreground',
      retro:
        'border-2 border-foreground bg-background shadow-[4px_4px_0px_0px_rgba(0,0,0,1)] dark:shadow-[4px_4px_0px_0px_rgba(255,255,255,1)] rounded-none text-foreground',
      glow: 'bg-card border-primary/20 shadow-[0_0_25px_rgba(168,85,247,0.08)] text-card-foreground',
    },
  },
  defaultVariants: {
    variant: 'default',
  },
})

const emptyIconVariants = tv({
  base: 'flex items-center justify-center rounded-2xl p-3.5 mb-4 text-muted-foreground transition-all duration-200',
  variants: {
    variant: {
      default: 'bg-muted border border-border/60',
      glass: 'bg-white/10 dark:bg-white/5 border border-white/15 backdrop-blur-md text-foreground',
      retro:
        'border-2 border-foreground bg-background shadow-[2px_2px_0px_0px_rgba(0,0,0,1)] dark:shadow-[2px_2px_0px_0px_rgba(255,255,255,1)] rounded-none text-foreground',
      glow: 'bg-primary/10 border border-primary/30 text-primary shadow-[0_0_15px_rgba(168,85,247,0.2)]',
    },
  },
  defaultVariants: {
    variant: 'default',
  },
})

export interface EmptyProps
  extends React.HTMLAttributes<HTMLDivElement>,
    VariantProps<typeof emptyVariants> {}

export const Empty = React.forwardRef<HTMLDivElement, EmptyProps>(
  ({ className, variant = 'default', children, ...props }, ref) => {
    return (
      <div ref={ref} className={cn(emptyVariants({ variant }), className)} {...props}>
        {children}
      </div>
    )
  },
)
Empty.displayName = 'Empty'

export interface EmptyIconProps
  extends React.HTMLAttributes<HTMLDivElement>,
    VariantProps<typeof emptyIconVariants> {}

export const EmptyIcon = React.forwardRef<HTMLDivElement, EmptyIconProps>(
  ({ className, variant = 'default', children, ...props }, ref) => {
    return (
      <div ref={ref} className={cn(emptyIconVariants({ variant }), className)} {...props}>
        {children}
      </div>
    )
  },
)
EmptyIcon.displayName = 'EmptyIcon'

export const EmptyMedia = EmptyIcon

export const EmptyTitle = React.forwardRef<HTMLHeadingElement, React.HTMLAttributes<HTMLHeadingElement>>(
  ({ className, children, ...props }, ref) => {
    return (
      <h3 ref={ref} className={cn('text-base font-semibold tracking-tight text-foreground mb-1', className)} {...props}>
        {children}
      </h3>
    )
  },
)
EmptyTitle.displayName = 'EmptyTitle'

export const EmptyDescription = React.forwardRef<HTMLDivElement, React.HTMLAttributes<HTMLDivElement>>(
  ({ className, children, ...props }, ref) => {
    return (
      <div ref={ref} className={cn('text-xs text-muted-foreground max-w-xs mb-6 leading-relaxed', className)} {...props}>
        {children}
      </div>
    )
  },
)
EmptyDescription.displayName = 'EmptyDescription'

export const EmptyActions = React.forwardRef<HTMLDivElement, React.HTMLAttributes<HTMLDivElement>>(
  ({ className, children, ...props }, ref) => {
    return (
      <div ref={ref} className={cn('flex items-center justify-center gap-2.5', className)} {...props}>
        {children}
      </div>
    )
  },
)
EmptyActions.displayName = 'EmptyActions'
```
