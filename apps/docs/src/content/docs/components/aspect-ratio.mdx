<Installation name="aspect-ratio">

```tsx
'use client'

import * as React from 'react'
import { cn } from '@/lib/utils'

export interface AspectRatioProps extends React.HTMLAttributes<HTMLDivElement> {
  ratio?: number
}

const AspectRatio = React.forwardRef<HTMLDivElement, AspectRatioProps>(
  ({ className, ratio = 16 / 9, style, ...props }, ref) => {
    return (
      <div
        ref={ref}
        style={
          {
            aspectRatio: `${ratio}`,
            ...style,
          } as React.CSSProperties
        }
        className={cn('w-full overflow-hidden relative', className)}
        {...props}
      />
    )
  },
)
AspectRatio.displayName = 'AspectRatio'

export { AspectRatio }
```

<PropsTable title="AspectRatioProps">
  <PropRow
    prop="ratio"
    type="number"
    default="16 / 9"
    description="The target aspect ratio for the container wrapper. Defaults to standard landscape widescreen."
  />
  <PropRow
    prop="className"
    type="string"
    default="undefined"

  />
  <PropRow
    prop="style"
